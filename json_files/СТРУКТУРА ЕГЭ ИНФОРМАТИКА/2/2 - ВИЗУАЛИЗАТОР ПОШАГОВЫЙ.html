from itertools import * Импортируем itertools: permutations для перестановок, product для наборов аргументов.
def fun(x, y, z, w): Определяем логическую функцию F(x, y, z, w).
return not( (x <= (not w)) and z ) and not(w <= z) and (x <= (not z)) Тело функции: три условия через and. Возвращает True/False.
for per in permutations('xyzw'): Перебираем все 24 перестановки переменных x, y, z, w — какая из них даст заданную таблицу.
for a, b, c, d, e in product((0, 1), repeat=5): Перебираем все 32 набора из 5 значений (a, b, c, d, e) — это заполнение пропусков в таблице.
tab = [(1, 0, a, 0, 1), (1, 0, b, c, 0), (d, 1, e, 1, 0)] Собираем три строки таблицы: в каждой — 4 аргумента и значение F.
rez = [] Список для значений F, вычисленных функцией fun.
for t in tab: Идём по каждой строке таблицы.
rez.append(fun(**dict(zip(per, t)))) Сопоставляем per (x,y,z,w) с значениями строки t и вычисляем fun. Добавляем результат в rez.
if len(set(tab)) == 3 and rez == [1, 0, 0]: Проверяем: все 3 строки разные (len(set(tab))==3) и F совпадает с таблицей (rez == [1,0,0]).
print(*per, tab) Печатаем найденную перестановку и таблицу.
break Прерываем цикл по product — нам нужен только первый подходящий вариант.
